<script>
  // ── Gateway routing ──────────────────────────────────
  const gatewayScreen = document.getElementById('gateway-screen');
  const searchScreen  = document.getElementById('search-screen');
  let searchJsLoaded  = false;

  document.getElementById('btn-onedrive').addEventListener('click', function () {
    window.location.href = 'prototype2.php';
  });

  document.getElementById('btn-gdrive').addEventListener('click', function () {
    gatewayScreen.classList.add('hidden');
    searchScreen.classList.add('visible');

    // Load search.js only once, only after user chooses Google Drive
    if (!searchJsLoaded) {
      searchJsLoaded = true;
      const s = document.createElement('script');
      s.src = 'assets/js/search.js';
      document.body.appendChild(s);
    }
  });

  document.getElementById('back-btn').addEventListener('click', function () {
    searchScreen.classList.remove('visible');
    gatewayScreen.classList.remove('hidden');
  });
</script>
